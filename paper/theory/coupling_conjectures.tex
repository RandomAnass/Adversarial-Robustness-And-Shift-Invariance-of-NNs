\author{Working note (theory worked out by hand; agents used only for literature/refutation checks)}

\begin{abstract}
This note isolates the ``self-limiting coupling'' theory behind the paper's
margin--sensitivity discussion. Earlier drafts stated four conjectures (A--D); this
version records what is now settled. The single controlling quantity is the
\emph{spherical log-margin gradient} $\|\grad_{\Sph}\log m(u)\|$, equivalently
$\tan\angle(\grad M(x),x)$: the upper envelope on $\|\grad M\|$ holds iff this is
bounded, and the same quantity controls the certificate's looseness. We give the
exact identity (\S\ref{sec:Aprime}), a sharp linear case, a conditional bound under
global max-margin, an explicit counterexample showing no universal constant exists,
the corrected lockstep statement (\S\ref{sec:Bprime}), the corrected local capacity
statement (\S\ref{sec:Cprime}), and the bracket-condition-number result
(\S\ref{sec:D}). The implicit-bias envelope at the gradient-descent solution is
closed (\S\ref{sec:open}) by replacing the global margin with the achieved normalized
margin $\gamma_{\mathrm{IB}}$, which recovers the global statement when the limit is
optimal and degrades gracefully to a KKT point otherwise. Everything labelled
Proposition/Theorem here is proved.
\end{abstract}

\section{Setup}\label{sec:setup}
Let $f:\R^{\dd}\to\R^{k}$ be a classifier with logits $f_1,\dots,f_k$ and, for a
labelled point $(x,y)$, signed margin $M(x)=f_y(x)-\max_{j\neq y}f_j(x)$, so $M>0$
iff $f$ is correct. Write $\eta=\E[M]$ over correct points and $L=\E\|\grad M\|$ for
the mean input-gradient norm (the local sensitivity; norm dual to the threat model).
The first-order certificate is $r_2\ge M(x)/\|\grad M(x)\|_2$
(Hein--Andriushchenko 2017; Tsuzuku et al.\ 2018), so $\eta/L$ is a robustness
surrogate. The empirical finding is that $\eta/L$ \emph{diagnoses} robustness across
architectures yet \emph{training} it does not buy robustness; the question is what,
precisely, is true behind that.

\section{The radial envelope (Euler)}\label{sec:radial}
\begin{proposition}[Homogeneity lower envelope]\label{prop:euler}
Let $f$ be bias-free with degree-one positively homogeneous activations (ReLU,
leaky-ReLU, absolute value, maxout). Then $M$ is degree-one positively homogeneous,
$\langle\grad M(x),x\rangle=M(x)$ (Euler) holds a.e., and hence
\[
  \|\grad M(x)\|_2\ \ge\ \frac{M(x)}{\|x\|_2},\qquad
  \|\grad M(x)\|_1\ \ge\ \frac{M(x)}{\|x\|_\infty},\qquad
  \eta/L\le\sup_i\|x_i\|_2 .
\]
\end{proposition}
This is exact and standard (Euler's identity plus Cauchy--Schwarz / H\"older). The
content is only the reading that the margin gradient has an architecture-imposed
\emph{radial} floor. It controls the radial component of the gradient and nothing
else, which is exactly the limitation made precise in \S\ref{sec:Aprime}.

\section{Why the ``impossibility'' reading of the coupling is wrong}\label{sec:wrong}
An earlier draft read Proposition~\ref{prop:euler} as forbidding the training of
$\eta/L$. It does not.
\begin{itemize}
\item[(W1)] \emph{Wrong direction.} To raise $\eta/L=M/\|\grad M\|$ one wants
$\|\grad M\|$ small. Proposition~\ref{prop:euler} is a \emph{lower} bound on
$\|\grad M\|$, i.e.\ an \emph{upper} cap $\eta/L\le\|x\|$; an upper cap above the
operating value cannot forbid increasing the ratio.
\item[(W2)] \emph{Loose at the operating point.} Empirically
$\|\grad M\|_2\approx 26\,M/\|x\|_2$, an order of magnitude of slack.
\item[(W3)] \emph{Hypotheses fail.} Euler needs bias-free, exactly degree-1
homogeneous; the trained models carry biases and batch normalization.
\item[(W4)] \emph{Static vs.\ dynamic.} The empirical obstruction is a property of
the training path, not of a static inequality.
\end{itemize}
What survives unconditionally is the scale degeneracy.
\begin{lemma}[Scale degeneracy of the ratio]\label{lem:scale}
$M(x)/\|\grad M(x)\|$ is invariant under $f\mapsto cf$ ($c>0$). Hence ``maximize
$\eta/L$'' (equivalently penalize $\|\grad M\|/M$) is blind to absolute margin and
admits the constant classifier $\grad M\equiv0$ as a global optimizer; direct ratio
maximization is ill-posed. (Tsuzuku et al.\ 2018, App.\ G.2.)
\end{lemma}

\section{The exact spherical envelope (resolves A)}\label{sec:Aprime}
The radial floor controls only the radial gradient. Writing the gradient in polar
form isolates the missing piece exactly.

\begin{proposition}[Radial--tangential split of the margin gradient]\label{prop:Aprime}
Let $M:\R^{\dd}\setminus\{0\}\to\R$ be $C^1$ and positively $1$-homogeneous. For
$x\neq0$ set $r=\|x\|_2$, $u=x/r$, and $m(u)=M(u)$, so $M(x)=r\,m(u)$. Let
$\grad_{\Sph}m(u)$ be the spherical gradient (the tangential part). Then
\[
  \grad M(x)=m(u)\,u+\grad_{\Sph}m(u),\qquad u\perp\grad_{\Sph}m(u),
\]
hence
\[
  \|\grad M(x)\|_2^2=\Big(\frac{M(x)}{\|x\|_2}\Big)^2+\|\grad_{\Sph}m(u)\|_2^2,
\]
and, when $M(x)>0$,
\[
  \boxed{\ \|\grad M(x)\|_2=\frac{M(x)}{\|x\|_2}\sqrt{1+\|\grad_{\Sph}\log m(u)\|_2^2}\ }.
\]
Consequently the two-sided envelope
$\frac{M(x_i)}{\|x_i\|_2}\le\|\grad M(x_i)\|_2\le\kappa\,\frac{M(x_i)}{\|x_i\|_2}$
holds at points $x_i$ \emph{iff} $\|\grad_{\Sph}\log m(u_i)\|_2\le\sqrt{\kappa^2-1}$,
with $u_i=x_i/\|x_i\|_2$. Equivalently $\kappa_i=\sec\angle(\grad M(x_i),x_i)$.
\end{proposition}
\begin{proof}
Decompose a perturbation $h=a\,u+z$ with $a=\langle h,u\rangle$, $z=h-au\perp u$.
Then $Dr(x)[h]=\langle u,h\rangle=a$ and $Du(x)[h]=(h-au)/r=z/r$ (tangent to $\Sph$).
From $M(x)=r\,m(u)$,
$DM(x)[h]=a\,m(u)+r\,Dm(u)[z/r]=a\,m(u)+\langle\grad_{\Sph}m(u),z\rangle$. Setting
$g=m(u)u+\grad_{\Sph}m(u)$ and using $\langle u,z\rangle=0$,
$\langle\grad_{\Sph}m(u),u\rangle=0$ gives $\langle g,h\rangle=DM(x)[h]$ for all $h$,
so $\grad M(x)=g$. Orthogonality of the two pieces gives the Pythagorean norm
identity; $m(u)=M(x)/\|x\|_2$ and
$\grad_{\Sph}\log m=\grad_{\Sph}m/m$ give the boxed form. The envelope equivalence is
immediate.
\end{proof}

\noindent\textbf{The controlling quantity.} Euler fixes the radial component
$m(u)=M/\|x\|$; the entire upper envelope is the \emph{tangential} component
$\|\grad_{\Sph}\log m(u)\|=\tan\angle(\grad M,x)$. ``Can the coupling be tightened''
is exactly ``is the spherical log-margin gradient bounded at the support
directions.'' Homogeneity alone gives no such bound (it constrains only the radial
part), so A is not provable from homogeneity; it requires an implicit-bias input.

\subsection{Sharp linear case}\label{sec:linear}
\begin{proposition}[Exact for linear max-margin]\label{prop:linear}
For binary labels $y_i\in\{\pm1\}$ and a linearly separable dataset, let
$M_w(x)=y\,w^\top x$ and let $w^\star=\arg\min\|w\|_2$ s.t.\ $y_iw^\top x_i\ge1$ be
the hard-margin solution, with geometric margin
$\gamma=\sup_{\|w\|_2\le1}\min_i y_iw^\top x_i$. Then $\|w^\star\|_2=1/\gamma$, and at
every support point ($y_i(w^\star)^\top x_i=1$),
\[
  \frac{\|\grad_x M_{w^\star}(x_i)\|_2}{M_{w^\star}(x_i)/\|x_i\|_2}
  =\frac{\|x_i\|_2}{\gamma},\qquad
  \|\grad_{\Sph}\log m(u_i)\|_2=\sqrt{\Big(\tfrac{\|x_i\|_2}{\gamma}\Big)^2-1}.
\]
So $\kappa_i=\|x_i\|_2/\gamma$ exactly: the envelope constant is the data
condition number (radius-to-margin), and it is sharp.
\end{proposition}
\begin{proof}
$\grad_x M_w=y_iw$ so $\|\grad_xM_w\|_2=\|w\|_2$. The norm--margin duality
$\|w^\star\|_2=1/\gamma$ is the standard rescaling argument (normalize any feasible
$w$ to unit norm to lower-bound $\gamma$; rescale a near-optimal unit $v$ to feasibility
to upper-bound). At a support point $M_{w^\star}(x_i)=1$, so
$\|\grad M\|_2/(M/\|x_i\|_2)=\|w^\star\|_2\|x_i\|_2=\|x_i\|_2/\gamma$; substitute into
Proposition~\ref{prop:Aprime}.
\end{proof}

\subsection{Conditional bound under global max-margin}\label{sec:Thm1}
\begin{theorem}[Complexity-controlled max-margin envelope]\label{thm:complexity}
Let the model be positively homogeneous of degree $D>0$ in its parameters,
$M_{a\theta}(x_i)=a^DM_\theta(x_i)$, and let $C(\theta)\ge0$ be a complexity with the
same homogeneity, $C(a\theta)=a^DC(\theta)$, that controls the local input gradient,
$\|\grad_x M_\theta(x_i)\|_2\le B_iC(\theta)$ for all $i$. Define
$\gamma_C=\sup_{C(\theta)\le1}\min_i M_\theta(x_i)\in(0,\infty)$ and let $\theta^\star$
solve $\min_\theta C(\theta)$ s.t.\ $M_\theta(x_i)\ge1$. Then $C(\theta^\star)=1/\gamma_C$
and, at every support point,
\[
  \|\grad_x M_{\theta^\star}(x_i)\|_2\le\frac{B_i}{\gamma_C}
  =\frac{B_i\|x_i\|_2}{\gamma_C}\cdot\frac{M_{\theta^\star}(x_i)}{\|x_i\|_2},
  \qquad\text{i.e. } \kappa_i\le\frac{B_i\|x_i\|_2}{\gamma_C}.
\]
If moreover $M_{\theta^\star}$ is $1$-homogeneous in $x$, then
$\|\grad_{\Sph}\log m(u_i)\|_2\le\sqrt{(B_i\|x_i\|_2/\gamma_C)^2-1}$.
\end{theorem}
\begin{proof}
$C(\theta^\star)=1/\gamma_C$ is the homogeneous norm--margin duality: any feasible
$\theta$ rescaled by $C(\theta)^{-1/D}$ has unit complexity and margins
$\ge C(\theta)^{-1}$, so $\gamma_C\ge C(\theta)^{-1}$, giving $C(\theta)\ge1/\gamma_C$;
conversely a near-optimal unit-complexity $\theta_\varepsilon$ rescaled to feasibility
has complexity $\le1/(\gamma_C-\varepsilon)$. Substitute $C(\theta^\star)=1/\gamma_C$
into the gradient-control hypothesis and use $M_{\theta^\star}(x_i)=1$ at support;
the spherical form follows from Proposition~\ref{prop:Aprime}.
\end{proof}

\begin{corollary}[Bias-free depth-$D$ ReLU networks]\label{cor:relu}
For a bias-free depth-$D$ network with $1$-Lipschitz, $1$-homogeneous activations and
$C(\theta)=(\|\theta\|_2/\sqrt D)^D$ (with $\|\theta\|_2^2=\sum_\ell\|W_\ell\|_F^2$),
the input-gradient bound holds with $B_i=\sqrt2$: at a differentiability point with a
unique rival class $j^\star$, $\grad_xM=J_f^\top(e_y-e_{j^\star})$, so
$\|\grad_xM\|_2\le\sqrt2\,\|J_f\|_{2\to2}\le\sqrt2\prod_\ell\|W_\ell\|_2
\le\sqrt2\prod_\ell\|W_\ell\|_F\le\sqrt2\,(\|\theta\|_2^2/D)^{D/2}=\sqrt2\,C(\theta)$,
the last step by AM--GM. Hence, at the min-norm (global max-margin) interpolant,
$\kappa_i\le\sqrt2\,\|x_i\|_2/\gamma_{\mathrm{net}}$.
\end{corollary}

\subsection{No universal constant (the counterexample)}\label{sec:counter}
\begin{proposition}[The envelope constant must depend on data geometry]\label{prop:counter}
There is no universal $C$ with $\|\grad_{\Sph}\log m(u_i)\|_2\le C$ at the max-margin
solution, even for linear classifiers. Fix $A\ge1$ in $\R^2$ and take
$(e_1,+1),(-e_1,-1),(e_2/A,+1),(-e_2/A,-1)$. The hard-margin solution is
$w^\star=(1,A)$ (constraints force $w_1\ge1$, $w_2\ge A$, minimized at equality), so at
$x_1=e_1$: $M=1$, $\grad_xM=(1,A)$, radial part $(1,0)$, tangential part $(0,A)$, hence
$\|\grad_{\Sph}\log m(e_1)\|_2=A$ and $\kappa=\sqrt{1+A^2}=\|x_1\|_2/\gamma\to\infty$.
\end{proposition}
This realizes the spherical blow-up at a genuine max-margin solution; agent~D
(\S\ref{sec:D}) exhibits the same blow-up for a bias-free degree-$1$ \emph{ReLU} net
($\kappa\approx21$ at maximal geometric margin), so the dependence on the data
condition number is real for the architecture, not an artifact of linearity.

\section{The lockstep, corrected (resolves B)}\label{sec:Bprime}
The conjectured inequality $\frac{d\eta}{d(\log L)}\ge c\eta$ is mis-signed for a
no-go: with $d\log L/d\lambda<0$ it only lower-bounds margin decay and permits
$d\log\eta/d\lambda=0$ (sensitivity drops at fixed margin), the opposite of an
obstruction. The correct statements:
\begin{proposition}[Exact lockstep on a scale orbit]\label{prop:Bprime}
Fix a margin function $\bar M$ on $x_1,\dots,x_n$ and let $M_\lambda=a(\lambda)\bar M$
with $a(\lambda)>0$. Then $\eta(\lambda)=a(\lambda)\bar\eta$,
$L(\lambda)=a(\lambda)\bar L$, so $\eta/L\equiv\bar\eta/\bar L$ is constant and, if
$a'\neq0$, $d\log\eta/d\log L=1$. A first-order penalty cannot change $\eta/L$ by mere
rescaling; it can only change it by changing the classifier's \emph{shape}.
\end{proposition}
\begin{proof}
$M_\lambda(x_i)=a(\lambda)\bar M_i$ and $\grad_xM_\lambda=a(\lambda)\grad_x\bar M$ give
$\eta(\lambda)=a\bar\eta$, $L(\lambda)=a\bar L$; the ratio cancels $a$ and the
log-derivatives coincide.
\end{proof}
\begin{proposition}[Implicit-function characterization; no lockstep in general]\label{prop:Bift}
Let $\theta_\lambda$ minimize $\mathcal L(\theta)+\lambda R(\theta)$ with
$R=\E\|\grad_xM\|_q$ and $H_\lambda=\grad^2\mathcal L+\lambda\grad^2R$. Then
$\dot\theta_\lambda=-H_\lambda^{-1}\grad R$, $dL/d\lambda=-\langle\grad R,H_\lambda^{-1}\grad R\rangle\le0$
(the penalty reduces sensitivity for free), and
$d(\eta/L)/d\lambda=-\langle\grad(\eta/L),H_\lambda^{-1}\grad R\rangle$ is an
alignment/operating-point quantity, not a fixed sign. In the linear case along the
ridge path $w(t)=(\Sigma+tI)^{-1}b$, $d\log\eta/d\log L=m_2^2/(m_1m_3)\le1$
($m_k=b^\top A^{-k}b$, $A=\Sigma+tI$; Cauchy--Schwarz), so $\eta/L$ is
\emph{non-decreasing}: the penalty raises the surrogate. At a ratio-critical
(robust-optimal) operating point $d(\eta/L)/d\lambda=0$ to first order, which is where
an adversarially-trained network sits; this is the honest explanation of the
four-intervention experiment. The reason raising the surrogate does not raise
robustness is that $\eta=\E[M]$ is a weak robustness surrogate (raising it can rotate
the classifier away from the robust-optimal direction).
\end{proposition}

\section{The local capacity ceiling, corrected (resolves C)}\label{sec:Cprime}
As stated C is false without noise: for noiseless $x_i\sim\mathcal N(0,I_d)$ with
$y_i=\operatorname{sign}(u^\top x_i)$, the interpolant $g(x)=u^\top x$ has $L=1$,
$\eta=\sqrt{2/\pi}$, so $\eta/L=\Theta(1)\gg\tilde O(\sqrt{p/(nd)})$; a constant
classifier gives $L=0$, $\eta/L=\infty$. (This matches Bubeck--Sellke 2021, who state
both that noise is necessary and that the bound needs the \emph{supremum}, not the
average, of $\|\grad f\|$.) The salvage is a local, noisy-pair statement.
\begin{proposition}[Local ceiling from close opposite-label pairs]\label{prop:Cprime}
Let $g$ be $C^1$ on a convex set, with signed margins $M_i=y_ig(x_i)>0$ and
$G_i=\|\grad g(x_i)\|_2$. Suppose the data form $m$ disjoint opposite-label pairs
$(a_s,b_s)$ at distances $\rho_s\le\rho_{\max}$, each segment in the domain, with the
endpoint-control condition $\int_0^1\|\grad g(\gamma_s(t))\|_2\,dt\le\frac C2(G_{a_s}+G_{b_s})$
for some $C\ge1$. With $\eta=\frac1{2m}\sum_s(M_{a_s}+M_{b_s})$,
$L=\frac1{2m}\sum_s(G_{a_s}+G_{b_s})$, then $\eta\le\frac{C\rho_{\max}}{2}L$, i.e.\
$\eta/L\le C\rho_{\max}/2$.
\end{proposition}
\begin{proof}
For a pair, $M_{a}+M_{b}=g(x_a)-g(x_b)=\int_0^1\langle\grad g(\gamma(t)),x_a-x_b\rangle dt
\le\rho\int_0^1\|\grad g(\gamma(t))\|_2dt\le\frac{C\rho}{2}(G_a+G_b)$ by FTC,
Cauchy--Schwarz, and endpoint control; sum over pairs and divide by $2m$.
\end{proof}
This is the honest local analogue of the law of robustness: a ceiling needs close
opposite-label forcing plus no gradient spikes on the connecting segment.

\section{The bracket condition number (resolves D)}\label{sec:D}
The certificate looseness is $\kappa_{\mathrm{br}}=L\,r_2/\eta\ge1$, with
$\kappa_{\mathrm{br}}=1$ iff $M$ is affine on the segment from $x$ to the boundary, so
$\kappa_{\mathrm{br}}$ measures failure of local linearity, a curvature quantity, not
a margin quantity. D as stated (a bound $g(\mu)$ in the separation margin alone) is
false: a bias-free degree-$1$ ReLU net can have $\kappa_{\mathrm{br}}\approx21$ at
\emph{maximal} geometric margin. Two true statements:
\begin{proposition}[Architectural bound; benign curvature]\label{prop:D}
\textup{(P1)} For any bias-free degree-$1$ homogeneous net,
$\kappa_{\mathrm{br}}\le\sec\angle(\grad_xM(x),x)$, the same envelope constant as
Proposition~\ref{prop:Aprime}; in particular the polynomial $t\mapsto t^{2n+1}$
(degree $>1$) blow-up is excluded ($\kappa=2n+1$ there comes from non-unit
homogeneity). \textup{(P2)} If $M$ has curvature $\beta$ along the path with
$2\beta\eta\le L^2$ then $\kappa_{\mathrm{br}}\le 2/(1+\sqrt{1-2\beta\eta/L^2})\le2$
(a CURE / local-linearity-regularization style bound).
\end{proposition}
\noindent P1 ties the bracket looseness to the \emph{same} spherical/angular
quantity as the envelope, so $\eta/L$ is loose exactly when the input gradient is
tangentially misaligned with the datum; the implicit bias does not certify the benign
regime (min-norm interpolants permit localized curvature spikes; this is why
curvature/local-linearity regularizers exist).

\section{The one controlling quantity, and the resolved implicit-bias gap}\label{sec:open}
\textbf{Synthesis.} A single quantity governs both the envelope and the certificate:
the spherical log-margin gradient $\|\grad_{\Sph}\log m(u_i)\|=\tan\angle(\grad M,x_i)$
at support directions, with $\sec\angle=\sqrt{1+\tan^2}$ equal to the envelope
$\kappa$ (Prop.~\ref{prop:Aprime}) and an upper bound on the bracket
$\kappa_{\mathrm{br}}$ (Prop.~\ref{prop:D}). It is exactly the data condition number
$\|x\|/\gamma$ for linear max-margin (Prop.~\ref{prop:linear}), bounded by
$B_i\|x\|/\gamma_C$ under global max-margin with gradient-controlling complexity
(Thm.~\ref{thm:complexity}, Cor.~\ref{cor:relu}), and unbounded across datasets
(Prop.~\ref{prop:counter}). The margin alone never bounds it.

\paragraph{The gap, closed by the achieved margin.} Theorem~\ref{thm:complexity} and
Corollary~\ref{cor:relu} were stated for the \emph{global} max-margin interpolant.
Gradient descent on homogeneous networks converges only to a \emph{KKT point} of the
max-margin program (Lyu--Li 2020, arXiv:1906.05890; Ji--Telgarsky 2020,
arXiv:2006.06657), which for ReLU networks need not be a local or global max-margin
optimum (Vardi--Shamir--Srebro 2022, arXiv:2110.02732); the linear case is the clean
benchmark, where GD converges to the hard-margin SVM direction (Soudry et al.\ 2018,
JMLR). A proof that depends only on ``GD reaches a KKT point'' therefore cannot deliver
the \emph{global}-margin bound. The fix is to replace the global margin $\gamma_C$ by
the normalized margin \emph{actually achieved} by the limiting direction,
$\gamma_{\mathrm{IB}}$; the envelope then holds for the implicit-bias output itself and
recovers the global statement when that output is globally optimal.

\begin{theorem}[Implicit-bias spherical envelope via the achieved margin]\label{thm:ib}
Let $x_1,\dots,x_n\in\R^{\dd}\setminus\{0\}$ and $M_\theta(x_i)$ be the signed margin.
Assume \textup{(i)} parameter homogeneity $M_{a\theta}(x_i)=a^pM_\theta(x_i)$ for some
$p>0$; \textup{(ii)} a matching homogeneous complexity $C(a\theta)=a^pC(\theta)$ with
$\|\grad_x M_\theta(x_i)\|_2\le B_iC(\theta)$ at differentiability points;
\textup{(iii)} the limiting gradient-descent direction $\bar\theta$ makes
$x\mapsto M_{\bar\theta}(x)$ positively $1$-homogeneous; and \textup{(iv)}
$\min_i M_{\bar\theta}(x_i)>0$. Set the achieved normalized margin
$\gamma_{\mathrm{IB}}:=\min_iM_{\bar\theta}(x_i)/C(\bar\theta)$ (scale-invariant) and
normalize $\widehat\theta:=(\min_iM_{\bar\theta}(x_i))^{-1/p}\bar\theta$, so
$\min_iM_{\widehat\theta}(x_i)=1$. Then at every support point ($M_{\widehat\theta}(x_i)=1$),
\[
\|\grad_x M_{\widehat\theta}(x_i)\|_2\le\frac{B_i}{\gamma_{\mathrm{IB}}}
=\frac{B_i\|x_i\|_2}{\gamma_{\mathrm{IB}}}\cdot\frac{M_{\widehat\theta}(x_i)}{\|x_i\|_2},
\qquad
\|\grad_{\Sph}\log m(u_i)\|_2\le\sqrt{\Big(\tfrac{B_i\|x_i\|_2}{\gamma_{\mathrm{IB}}}\Big)^2-1}.
\]
If $B_i\le B$ and $\|x_i\|_2\le R$ on the support set, $\kappa_i\le BR/\gamma_{\mathrm{IB}}$.
\end{theorem}
\begin{proof}
$\gamma_{\mathrm{IB}}$ is scale-invariant: rescaling $\bar\theta\mapsto a\bar\theta$
multiplies both $\min_iM_{\bar\theta}(x_i)$ and $C(\bar\theta)$ by $a^p$. With
$\alpha=(\min_iM_{\bar\theta}(x_i))^{-1/p}$, parameter homogeneity gives
$M_{\widehat\theta}(x_i)=\alpha^pM_{\bar\theta}(x_i)=M_{\bar\theta}(x_i)/\min_jM_{\bar\theta}(x_j)$,
so $\min_iM_{\widehat\theta}(x_i)=1$, and $C(\widehat\theta)=\alpha^pC(\bar\theta)
=C(\bar\theta)/\min_iM_{\bar\theta}(x_i)=1/\gamma_{\mathrm{IB}}$. The gradient-control
hypothesis then gives $\|\grad_x M_{\widehat\theta}(x_i)\|_2\le B_iC(\widehat\theta)=B_i/\gamma_{\mathrm{IB}}$;
at a support point $M_{\widehat\theta}(x_i)=1$ rewrites this in the displayed form. The
spherical version follows by dividing by $m(u_i)=M_{\widehat\theta}(x_i)/\|x_i\|_2$ in
Proposition~\ref{prop:Aprime}.
\end{proof}
\begin{corollary}[Bias-free ReLU networks]\label{cor:ib-relu}
For a depth-$D$ bias-free network with $1$-Lipschitz $1$-homogeneous activations and
$C(\theta)=(\|\theta\|_2/\sqrt D)^D$, the gradient-control hypothesis holds with
$B_i=\sqrt2$ (Corollary~\ref{cor:relu}), so the implicit-bias direction satisfies
$\|\grad_x M_{\widehat\theta}(x_i)\|_2\le\sqrt2/\gamma_{\mathrm{IB}}$ and
$\|\grad_{\Sph}\log m(u_i)\|_2\le\sqrt{2\|x_i\|_2^2/\gamma_{\mathrm{IB}}^2-1}$ at support points.
\end{corollary}
\begin{remark}[Why this is the right denominator]
When the gradient-descent limit is globally optimal, $\gamma_{\mathrm{IB}}=\gamma_C$ and
Theorem~\ref{thm:ib} reduces to Theorem~\ref{thm:complexity}. When it is only a KKT
point, the achieved margin $\gamma_{\mathrm{IB}}\le\gamma_C$ is smaller and the bound
weakens accordingly, which is exactly the behaviour forced by the Vardi--Shamir--Srebro
counterexamples: KKT convergence does not license replacing $\gamma_{\mathrm{IB}}$ by
$\gamma_C$. The linear case is exact ($\|\grad_xM_w\|_2=\|w\|_2$, $C=\|w\|_2$, equality),
and Proposition~\ref{prop:counter} shows no denominator weaker than the
solution's data condition number can work. Open residuals: comparing
$\gamma_{\mathrm{IB}}$ to $\gamma_C$ (when GD is globally optimal) is the standard hard
implicit-bias question; B in the genuine non-scale nonlinear regime; C in the noisy
$p\ll n\dd$ training-point-average niche; D's explicit GD dataset realizing large
$\kappa_{\mathrm{br}}$ at large margin.
\end{remark}

\section{Orbit-complete interpolation has no invariant norm gap (corrects E1 / the lazy-cluster theorem)}
\label{sec:no-gap}
The lazy-regime projection argument and its rich-regime variation-norm analogue must be stated
with care. If the training set is closed under the group action and labels are constant on orbits,
enforcing invariance \emph{cannot} reduce the minimum interpolation norm: the invariant class is a
subset of the unconstrained class, while Reynolds averaging maps every feasible unconstrained
interpolant to an invariant feasible interpolant of no larger norm, so the two optimal values
coincide. Consequently a strict certified-Lipschitz gap cannot come from the same norm certificate
on orbit-complete data. \textbf{This corrects the draft's lazy-cluster theorem}, which claims a
strict gap equal to the orbit-variance $\|(I-\Pi_G)f_{\mathrm{unc}}\|^2$: for a strictly convex
norm (the RKHS case) the unconstrained minimum-norm interpolant on orbit-complete data is
\emph{already invariant}, so that orbit-variance is exactly zero.

\begin{theorem}[No invariant norm gap on orbit-complete data]\label{thm:no-gap}
Let a compact group $G$ act on the inputs, with induced action $(T_gf)(x)=f(g^{-1}x)$. Let
$(\mathcal F,\|\cdot\|_{\mathcal F})$ be a convex class with a $G$-invariant convex norm
($\|T_gf\|_{\mathcal F}=\|f\|_{\mathcal F}$), let $\mu$ be normalized Haar measure, and let the
Reynolds operator $\Pi_Gf:=\int_G T_gf\,d\mu(g)$ satisfy $\Pi_Gf\in\mathcal F$ for $f\in\mathcal F$.
Let the data $\{(x_i,y_i)\}_{i=1}^n$, $y_i\in\{\pm1\}$, be \emph{orbit-complete and orbit-constant}:
for every $i$ and $g\in G$, $gx_i$ is a training point with label $y_i$. With
$\mathcal F^G=\{f\in\mathcal F:T_gf=f\ \forall g\}$, define the unit-margin minimum-norm values
\[
P_{\mathrm{unc}}=\inf_{f\in\mathcal F}\{\|f\|_{\mathcal F}: y_if(x_i)\ge1\ \forall i\},\qquad
P_{\mathrm{inv}}=\inf_{f\in\mathcal F^G}\{\|f\|_{\mathcal F}: y_if(x_i)\ge1\ \forall i\}.
\]
Then $P_{\mathrm{unc}}=P_{\mathrm{inv}}$. If an unconstrained minimizer $f_\star$ exists, $\Pi_Gf_\star$
is an invariant minimizer; if $\|\cdot\|_{\mathcal F}$ is strictly convex, every unconstrained
minimizer is already invariant.
\end{theorem}
\begin{proof}
\emph{(1) $\Pi_G$ is non-expansive.} By Jensen and norm-invariance,
$\|\Pi_Gf\|\le\int_G\|T_gf\|\,d\mu=\|f\|\int_G d\mu=\|f\|$ (Haar normalized).
\emph{(2) $\Pi_G$ preserves feasibility and lands in $\mathcal F^G$.} If $y_if(x_i)\ge1$ for all $i$,
then for each $g$, $g^{-1}x_i$ is a training point with label $y_i$, so $y_i(T_gf)(x_i)=y_if(g^{-1}x_i)\ge1$;
averaging over $g$ gives $y_i(\Pi_Gf)(x_i)=\int_G y_i(T_gf)(x_i)\,d\mu\ge1$. Invariance:
$T_h\Pi_Gf=\int_G T_{hg}f\,d\mu(g)=\int_G T_kf\,d\mu(k)=\Pi_Gf$ by left-invariance of Haar.
\emph{(3) Equal values.} $\mathcal F^G\subseteq\mathcal F$ gives $P_{\mathrm{unc}}\le P_{\mathrm{inv}}$.
For the reverse, take $f_\varepsilon$ feasible with $\|f_\varepsilon\|\le P_{\mathrm{unc}}+\varepsilon$;
by (2) $\Pi_Gf_\varepsilon$ is invariant feasible and by (1) $\|\Pi_Gf_\varepsilon\|\le P_{\mathrm{unc}}+\varepsilon$,
so $P_{\mathrm{inv}}\le P_{\mathrm{unc}}+\varepsilon$; let $\varepsilon\downarrow0$.
\emph{(4) Self-symmetrization.} If $f_\star$ minimizes, $\Pi_Gf_\star$ is invariant feasible with
$\|\Pi_Gf_\star\|\le\|f_\star\|=P_{\mathrm{unc}}=P_{\mathrm{inv}}$, hence an invariant minimizer. If the
norm is strictly convex and some $f_\star$ had $T_hf_\star\ne f_\star$: both are feasible minimizers of
equal norm (each $T_gf_\star$ is feasible since $g^{-1}x_i$ is a same-label training point), so their
average is feasible with strictly smaller norm, contradicting optimality; hence $f_\star\in\mathcal F^G$.
\end{proof}

\begin{corollary}[Variation-norm / rich two-layer]\label{cor:no-gap-var}
For an infinite-width two-layer network $f(x)=\int a\,\sigma(w^\top x+b)\,d\rho$ with variation
(atomic) norm $\|f\|_{\mathcal V}=\inf_\rho\int|a|\,d|\rho|$, if the activation dictionary is closed
under the (orthogonal) $G$-action---for each atom $\phi$ and $g$, $T_g\phi$ is again an atom---then
$\|\cdot\|_{\mathcal V}$ is $G$-invariant and Theorem~\ref{thm:no-gap} applies: on orbit-complete data
the unconstrained and invariant minimum-variation values coincide. (The norm is not strictly convex,
so an invariant optimum \emph{exists} but need not be unique.)
\end{corollary}

\begin{corollary}[RKHS: orbit-complete data force invariant interpolation]\label{cor:no-gap-rkhs}
For an RKHS $\mathcal H_K$ with a $G$-invariant kernel, the action is an isometry and the norm is
strictly convex, so on orbit-complete data the unit-margin minimum-norm interpolant is unique and
$G$-invariant. With a certificate $L_{\mathrm{cert}}(f)=L_\Phi\|f\|_K$, equal minimizing norms give
equal best certified constants: there is \emph{no} strict RKHS-norm or certified-Lipschitz gap from
the projection argument on orbit-complete data.
\end{corollary}

\begin{remark}[What survives, and the correct open problem]\label{rem:no-gap}
The theorem does not say invariance cannot help robustness; only that on orbit-complete data the help
cannot be a smaller value of the \emph{same} minimum norm. Three routes remain. (i) On
\emph{non}-orbit-complete data, $\Pi_G$ need not preserve the margin constraints, so the invariant and
unconstrained problems genuinely differ. (ii) Even at equal minimum norm, the \emph{realized} local
gradient $\E_i\|\nabla_x f(x_i)\|$ is not a norm value and may differ among equal-norm interpolants---
this is where a realized $\eta/L$ advantage lives. (iii) For the non-strictly-convex variation norm the
optimum is non-unique, so the rich-regime question is correctly localized: not a smaller variation-norm
optimum, but whether the \emph{dynamics select} an invariant / lower-local-gradient optimum among
equal-norm interpolants. Two consequences for the paper. First, the empirical lazy-regime cluster
advantage (and the synthetic-data smoothness effect) is a \emph{realized}-local-Lipschitz / finite-width
trajectory phenomenon, not an infinite-width min-norm gap. Second, on orbit-complete data with a
strictly convex (RKHS) norm there is no advantage at all in the exact limit, so the lazy theorem must be
restated as no-gap plus realized-Lipschitz localization (this is the corrected E1).
\end{remark}
